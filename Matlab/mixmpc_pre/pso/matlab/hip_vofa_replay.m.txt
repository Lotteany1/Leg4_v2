function hip_vofa_replay(out, baseline_only, tag)
p=tinympc14_config();
helper=fileparts(mfilename('fullpath'));
addpath(helper,'-begin');
assert(strcmpi(which('tinympc14_model_at'),fullfile(helper,'tinympc14_model_at.m')));
p.PSO_EXACT=true;
p.ONLINE_TERMINAL=[];
base=struct('id','baseline','Q_DIAG',p.Q_DIAG,'R_DIAG',p.R_DIAG);
rows=base;
if ~baseline_only
    file='ppo_results.json';
    if strcmp(tag,'local'), file='ppo_local_results.json'; end
    search=jsondecode(fileread(fullfile(fileparts(out),file)));
    [~,ranked]=sort([search.rows(2:end).train]);
    feasible=[search.rows(ranked+1).train_mid_guard];
    if strcmp(tag,'local'), feasible=feasible & [search.rows(ranked+1).heldout_mid_guard]; end
    [~,admissible]=sort(~feasible);
    rows=search.rows([1,ranked(admissible(1:3))+1]);
    assert(norm(rows(1).Q_DIAG(:)-p.Q_DIAG(:),inf)<1e-9 && norm(rows(1).R_DIAG(:)-p.R_DIAG(:),inf)<1e-9, ...
        'PPO experiment baseline differs from isolated MATLAB Q/R');
end
% Height, damping, speed reference, parked displacement, wheel/hip
% command delay in 2-ms ticks, added load (kg), and pitch disturbance.
scenes=[.17 0 -.52 0 3 .05 -.09;
        .18 .9 .61 0 4 .20 .11;
        .19 2.1 -.71 0 6 .15 -.10;
        .17 5.4 .63 0 9 .25 .13;
        .18 7.5 0 -.10 5 .10 -.12;
        .19 1.6 0 .11 4 .15 .12;
        .25 .6 .59 0 4 .20 .10;
        .26 2.3 -.72 0 6 .20 -.12;
        .24 5.0 0 -.10 7 .10 -.10;
        .25 0 0 .12 3 .15 .09];
for index=1:numel(rows)
    row=rows(index);
    path=fullfile(out,[char(row.id) '.json']);
    if isfile(path), continue; end
    config=p;
    config.Q_DIAG=row.Q_DIAG(:)';
    config.R_DIAG=row.R_DIAG(:)';
    clear metrics
    for k=1:size(scenes,1)
        scene=scenes(k,:);
        config.REPLAY_DAMP=scene(2);
        config.REPLAY_DELAY=scene(5);
        conditions=[scene(1) 0 4 9 .578 scene(6) .05 scene(3) 0 4 scene(4) 0 scene(7) 1];
        m=simulate(config,sprintf('replay_%02d',k),conditions);
        m.damping=scene(2);
        metrics(k)=m; %#ok<AGROW>
        fprintf('%s %d finite=%d stop=%.3f late_speed=%.4f attitude=%.4f\n',char(row.id),k, ...
            m.finite,m.speed_stop_s,m.late_speed_rms,m.attitude_peak);
    end
    row.mpc=metrics;
    write_json(path,row);
end
end
